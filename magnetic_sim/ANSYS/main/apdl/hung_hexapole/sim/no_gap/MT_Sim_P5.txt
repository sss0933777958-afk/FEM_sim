/PREP7

! ---- coil 5 = +1A raw, others 0（線圈幾何同 mesh deck：THK 3mm, H 7mm, N70）----
COIL_THK = 3e-3

R,4,1,TURNS*0,COIL_THK,COIL_H

R,6,1,TURNS*0,COIL_THK,COIL_H
R,5,1,TURNS*1,COIL_THK,COIL_H

! ---- BC: 外球 R180 面 MAG=0（DSP 唯一解）----
LOCAL,11,2,0,0,0
CSYS,11
NSEL,S,LOC,X,179e-3,181e-3
CSYS,0
D,ALL,MAG,0
ALLSEL,ALL
FINISH

! ---- solve (DSP scalar potential) ----
/SOLU
magsolv,3,,,,,1
FINISH
