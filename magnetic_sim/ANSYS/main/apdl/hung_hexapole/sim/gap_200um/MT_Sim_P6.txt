! ---- coil 6 = +1A raw, others 0（線圈幾何同 mesh deck：THK 3mm, H 7mm, N70）----

! ---- BC: 外球 R180 面 MAG=0（DSP 唯一解）----
LOCAL,11,2,0,0,0
CSYS,11
NSEL,S,LOC,X,179e-3,181e-3
CSYS,0
D,ALL,MAG,0
ALLSEL,ALL
FINISH

! ---- solve (DSP scalar potential) ----
/SOLU
magsolv,3,,,,,1
FINISH

NLIST,ALL, , ,COORD
/OUTPUT
/OUTPUT,'coil6_bfield_wp','dat','G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\hung_hexapole\data\coil6\gap_200um'
