! MT_Hung_SphereModel.txt
! [ADDED] 從球面幾何建立 6 根 D 型 pole
! 上層 (P2, P4, P5)：相對水平往上傾斜 35°
! 下層 (P1, P3, P6)：相對水平往下傾斜 5.71°
! R_sphere = 0.5mm, 方位角 120° 等距, magic angle
! Pole 尺寸來自 Mag_Pole_Bottom.STEP
! =======================================

/CWD,'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\hung_hexapole\db\mesh\mesh_graded_basegap'

/PREP7
/UNITS, MKS
/FILNAME,mesh_graded,1
*afun, DEG
BTOL, 1e-6
! [MODIFIED] SolidWorks 讀 IGES 固定當英寸 → 內部用英寸建模
! SolidWorks 匯入後自動 ×25.4 → 顯示為正確 mm 值
MM = 1e-3                     ! [MESH] 公尺 MKS
! [MODIFIED 2026-07-09 EXPERIMENT] 下極 gap slab radial：40=移進 Part1 series 鋼(導柱 43.2 內側、過縮短鐵棒 ~38.5)翻反增益；原 45/48 在 Part2 漏磁。上極不動 44.43
GLOW_R = 40.0
! [ADDED 2026-07-09 EXPERIMENT] 下極鐵棒長(只縮下極,末端~38.5<gap 40,block/導柱/coil 仍用 43)；必須 *MM 與 POLE_TOTAL_LEN 同單位(用作 fc_ex=p1+fc_rodlen*unitdir)。非 CAD baseline
ROD_LEN_LO = 38.0*MM
! [ADDED 2026-07-09 EXPERIMENT] 下極氣隙切刀 tangential 半寬 [mm]；5=10mm寬，配下極 block 高 10mm → 氣隙截面~100mm²(中央,兩側各留6mm鋼橋,不改鐵/partial cut)。上極切刀不動(±12)
GAP_HW_LO = 5.0

! ===== Pole 尺寸 (from Mag_Pole_Bottom.STEP) =====
POLE_R         = 3.175*MM     ! [MODIFIED] 圓柱半徑
POLE_CONE_LEN  = 15.875*MM    ! [MODIFIED] 錐段長度
POLE_TOTAL_LEN = 43.0*MM      ! [MODIFIED] 總長
POLE_FLAT_LEN  = 28.0*MM      ! [MODIFIED] 削平長度

! [MODIFIED] Tip fillet 80 µm diameter = 40 µm radius (與 sim 腳本 + Long 2016 一致)
POLE_TIP_R    = 0.040*MM
PCH           = sqrt(POLE_CONE_LEN*POLE_CONE_LEN + POLE_R*POLE_R)
FANG1         = acos(POLE_R/PCH)
FANG2         = 90 - FANG1
LFIL          = POLE_CONE_LEN*POLE_TIP_R/POLE_R
PTC           = LFIL/cos(FANG2)
PCN2          = LFIL*cos(FANG2)
PTR2          = POLE_TIP_R*cos(FANG2)
FDROP         = PTC - POLE_TIP_R
FPROT         = PCN2 - FDROP
JUNC_X        = POLE_CONE_LEN - FDROP    ! cone-cyl 接點 X (was POLE_CONE_LEN)

! ===== 球面幾何 =====
R_sphere = 0.5*MM             ! [MODIFIED] 球半徑
fc_theta = 54.74              ! [ADDED] magic angle (極角)
fc_h = cos(fc_theta)*R_sphere      ! [MODIFIED] 上層平面高度 (乘號前後不能有空格!)
fc_r = sin(fc_theta)*R_sphere      ! [MODIFIED] 上層圓半徑

TILT_UP = 35.0                ! [ADDED] 上層傾斜角 (度)
TILT_DN = 5.71                ! [ADDED] 下層傾斜角 (度)
BLK_YOKE_GAP = 14.0*MM

! ===== 6 根 pole 定義 =====
! [MODIFIED] ANSYS 座標對齊 measured 座標系：ANSYS +X = X_m = P1 方向
! 索引 1-6 = P1-P6
! 逆時針：P1(0°) P5(60°) P3(120°) P2(180°) P6(240°) P4(300°)
*dim, fc_tx, , 6
*dim, fc_ty, , 6
*dim, fc_tz, , 6
*dim, fc_ia, , 6              ! incline (正=上, 負=下)
*dim, fc_ra, , 6              ! azimuth (outward 方向, = measured 方位角)

! --- P1 (lower, 0°) ---
fc_tx(1) = fc_r
fc_ty(1) = 0
fc_tz(1) = -fc_h
fc_ia(1) = -TILT_DN
fc_ra(1) = 0

! --- P5 (upper, 60°) ---
fc_tx(5) = cos(60)*fc_r
fc_ty(5) = sin(60)*fc_r
fc_tz(5) = fc_h
fc_ia(5) = TILT_UP
fc_ra(5) = 60

! --- P3 (lower, 120°) ---
fc_tx(3) = cos(120)*fc_r
fc_ty(3) = sin(120)*fc_r
fc_tz(3) = -fc_h
fc_ia(3) = -TILT_DN
fc_ra(3) = 120

! --- P2 (upper, 180°) ---
fc_tx(2) = cos(180)*fc_r
fc_ty(2) = sin(180)*fc_r
fc_tz(2) = fc_h
fc_ia(2) = TILT_UP
fc_ra(2) = 180

! --- P6 (lower, 240°) ---
fc_tx(6) = cos(240)*fc_r
fc_ty(6) = sin(240)*fc_r
fc_tz(6) = -fc_h
fc_ia(6) = -TILT_DN
fc_ra(6) = 240

! --- P4 (upper, 300°) ---
fc_tx(4) = cos(300)*fc_r
fc_ty(4) = sin(300)*fc_r
fc_tz(4) = fc_h
fc_ia(4) = TILT_UP
fc_ra(4) = 300

! ===== 建立 6 根 pole（兩段式：前段 D 型 + 後段全圓）=====
*do, fc_pi, 1, 6, 1

    fc_incl = fc_ia(fc_pi)
    fc_azim = fc_ra(fc_pi)

    ! [ADDED 2026-07-09 EXPERIMENT] 每極鐵棒長：下極縮短(ROD_LEN_LO)使 gap(40) 前結束、不橋接；上極維持 POLE_TOTAL_LEN
    fc_rodlen = POLE_TOTAL_LEN
    *if, fc_incl, LT, 0, THEN
        fc_rodlen = ROD_LEN_LO
    *endif

    ! --- 方向向量 ---
    fc_ax = cos(fc_incl)*cos(fc_azim)
    fc_ay = cos(fc_incl)*sin(fc_azim)
    fc_az = sin(fc_incl)

    ! [MODIFIED] 截面偏移方向：決定削平面朝向
    ! Upper pole: -side_perp → VROTAT 180° 保留下半 → 削平面朝上
    ! Lower pole: +side_perp → VROTAT 180° 保留上半 → 削平面朝下（轉 180°）
    *if, fc_incl, GE, 0, THEN
        fc_nsx = -sin(fc_azim)
        fc_nsy = cos(fc_azim)
    *else
        fc_nsx = sin(fc_azim)
        fc_nsy = -cos(fc_azim)
    *endif

    ! --- Numbering ---
    *get, KP_NUM, KP, , NUM, MAX
    *if, KP_NUM, LT, 1, THEN
        KP_NUM = 0
    *endif
    KP_START = KP_NUM + 1
    NUMSTR, KP, KP_START

    *get, LI_NUM, LINE, , NUM, MAX
    *if, LI_NUM, LT, 1, THEN
        LI_NUM = 0
    *endif
    LI_START = LI_NUM + 1
    NUMSTR, LINE, LI_START

    ! --- Tip 位置 ---
    p1x = fc_tx(fc_pi)
    p1y = fc_ty(fc_pi)
    p1z = fc_tz(fc_pi)

    ! --- 軸上關鍵位置 ---
    ! 削平交界（tip 起 28mm）
    fc_fx = p1x + POLE_FLAT_LEN*fc_ax
    fc_fy = p1y + POLE_FLAT_LEN*fc_ay
    fc_fz = p1z + POLE_FLAT_LEN*fc_az
    ! 後端（tip 起 fc_rodlen；下極 38mm / 上極 43mm）[MODIFIED 2026-07-09 EXPERIMENT]
    fc_ex = p1x + fc_rodlen*fc_ax
    fc_ey = p1y + fc_rodlen*fc_ay
    fc_ez = p1z + fc_rodlen*fc_az

    ! ==========================================
    ! [MODIFIED] 前段：tip → 28mm（D 型，VROTAT 180°）+ 40µm fillet
    ! ==========================================
    ! KP+0: foremost (= 原 tip 位置, fillet 最前端)
    ! KP+1: 切線點 (cone 與 fillet 相切點，偏移 -side_perp)
    ! KP+2: 錐-柱交界 (用 JUNC_X，偏移 -side_perp)
    ! KP+3: 削平交界（偏移 -side_perp）
    ! KP+4: 削平交界（軸上）
    ! KP+5: 後端（軸上，僅作 VROTAT 軸用）
    ! KP+6: fillet 球心 (沿 +pole_axis 距 foremost = POLE_TIP_R)

    K, KP_START,   p1x, p1y, p1z
    K, KP_START+1, p1x + FPROT*fc_ax + PTR2*fc_nsx, p1y + FPROT*fc_ay + PTR2*fc_nsy, p1z + FPROT*fc_az
    K, KP_START+2, p1x + JUNC_X*fc_ax + POLE_R*fc_nsx, p1y + JUNC_X*fc_ay + POLE_R*fc_nsy, p1z + JUNC_X*fc_az
    K, KP_START+3, fc_fx + POLE_R*fc_nsx, fc_fy + POLE_R*fc_nsy, fc_fz
    K, KP_START+4, fc_fx, fc_fy, fc_fz
    K, KP_START+5, fc_ex, fc_ey, fc_ez
    K, KP_START+6, p1x + POLE_TIP_R*fc_ax, p1y + POLE_TIP_R*fc_ay, p1z + POLE_TIP_R*fc_az

    LARC, KP_START, KP_START+1, KP_START+6, POLE_TIP_R
    L, KP_START+1, KP_START+2
    L, KP_START+2, KP_START+3
    L, KP_START+3, KP_START+4
    L, KP_START+4, KP_START

    *get, AR_NUM, AREA, , NUM, MAX
    *if, AR_NUM, LT, 1, THEN
        AR_NUM = 0
    *endif
    AR_START = AR_NUM + 1
    NUMSTR, AREA, AR_START
    AL, LI_START, LI_START+1, LI_START+2, LI_START+3, LI_START+4

    VROTAT, AR_START, ,,,, , KP_START, KP_START+5, 180, 2

    ALLSEL, ALL

    ! ==========================================
    ! [ADDED] 後段：28mm → 43mm（全圓，VROTAT 360°）
    ! 用獨立 KP 避免共用問題
    ! ==========================================
    *get, KP_NUM, KP, , NUM, MAX
    KP_START2 = KP_NUM + 1
    NUMSTR, KP, KP_START2

    *get, LI_NUM, LINE, , NUM, MAX
    LI_START2 = LI_NUM + 1
    NUMSTR, LINE, LI_START2

    ! KP+0: 削平交界（軸上）— 獨立 KP
    ! KP+1: 削平交界（偏移 -side_perp）— 獨立 KP
    ! KP+2: 後端（偏移 -side_perp）
    ! KP+3: 後端（軸上）— 獨立 KP

    K, KP_START2,   fc_fx, fc_fy, fc_fz
    K, KP_START2+1, fc_fx + POLE_R*fc_nsx, fc_fy + POLE_R*fc_nsy, fc_fz
    K, KP_START2+2, fc_ex + POLE_R*fc_nsx, fc_ey + POLE_R*fc_nsy, fc_ez
    K, KP_START2+3, fc_ex, fc_ey, fc_ez

    L, KP_START2, KP_START2+1
    L, KP_START2+1, KP_START2+2
    L, KP_START2+2, KP_START2+3
    L, KP_START2+3, KP_START2

    *get, AR_NUM, AREA, , NUM, MAX
    AR_START2 = AR_NUM + 1
    NUMSTR, AREA, AR_START2
    AL, LI_START2, LI_START2+1, LI_START2+2, LI_START2+3

    VROTAT, AR_START2, ,,,, , KP_START2, KP_START2+3, , 4

    ALLSEL, ALL

    *msg, INFO, fc_pi
     Pole P%I built (D-shape + round)

*enddo

! ===== [ADDED] 安裝 block =====
! Upper block: 25×22×10mm, L型截面, 伸入7mm
! Lower block: 22×22×10mm, L型截面, 伸入4.5mm
! 寬邊朝 workspace, block 厚度沿 pole 軸

PENE_UP = 7.0*MM              ! 上層伸入量
PENE_DN = 4.5*MM              ! 下層伸入量
BLK_T   = 10.0*MM             ! block 厚度 (兩者相同)

*do, fc_pi, 1, 6, 1

    fc_incl = fc_ia(fc_pi)
    fc_azim = fc_ra(fc_pi)

    ! 軸方向
    fc_ax = cos(fc_incl)*cos(fc_azim)
    fc_ay = cos(fc_incl)*sin(fc_azim)
    fc_az = sin(fc_incl)

    ! up_perp (垂直軸,朝上)
    fc_ux = -sin(fc_incl)*cos(fc_azim)
    fc_uy = -sin(fc_incl)*sin(fc_azim)
    fc_uz = cos(fc_incl)

    ! side_perp (側向)
    fc_sx = sin(fc_azim)
    fc_sy = -cos(fc_azim)

    ! Tip 位置
    p1x = fc_tx(fc_pi)
    p1y = fc_ty(fc_pi)
    p1z = fc_tz(fc_pi)

    ! 判斷上下層 → 決定伸入量和 block 尺寸
    *if, fc_incl, GE, 0, THEN
        ! Upper pole → upper block (25×22×10)
        fc_pene = PENE_UP
        ! L型截面 (STEP coords → local: STEP_X=side, STEP_Y=up)
        ! Part1: side -0.5~12.5, up -11~11 (main body)
        ! Part2: side -12.5~-0.5, up -5~5 (extension)
        ! [MODIFIED] 平移 +0.25mm 讓 slot 中心對齊 pole 軸
        fc_s1a = -0.25*MM
        fc_s1b = 12.75*MM
        fc_u1a = -11.0*MM
        fc_u1b = 11.0*MM
        fc_s2a = -12.25*MM
        fc_s2b = -0.25*MM
        fc_u2a = -5.0*MM
        fc_u2b = 5.0*MM
    *else
        ! Lower pole → lower block (22×22×10)
        ! Part1: side 0~11, up -11~11 (main body)
        ! Part2: side -11~0, up -5~5 (extension)
        fc_pene = PENE_DN
        ! [MODIFIED] 平移 -2mm 讓 slot 中心(x=2)對齊 pole 軸
        fc_s1a = 2.0*MM       ! 4-2=2
        fc_s1b = 9.0*MM       ! 11-2=9
        fc_u1a = -11.0*MM
        fc_u1b = 11.0*MM
        fc_s2a = -13.0*MM     ! -11-2=-13
        fc_s2b = 2.0*MM       ! 4-2=2
        fc_u2a = -5.0*MM
        fc_u2b = 5.0*MM
    *endif

    ! Block 起點（沿 pole 軸，從 tip 算）
    fc_bstart = POLE_TOTAL_LEN - fc_pene

    ! [MODIFIED] Block 水平, global 座標, 不用 LOCAL CS
    ! Pole 尾端
    fc_endx = p1x + POLE_TOTAL_LEN*fc_ax
    fc_endy = p1y + POLE_TOTAL_LEN*fc_ay
    fc_endz = p1z + POLE_TOTAL_LEN*fc_az

    ! Entry point (workspace face, pole 尾端往回 PENE)
    fc_entx = fc_endx - fc_pene*fc_ax
    fc_enty = fc_endy - fc_pene*fc_ay
    fc_entz = fc_endz - fc_pene*fc_az

    ! Block Z 範圍 + XY 中心
    *if, fc_incl, GE, 0, THEN
        ! Upper: entry=bottom, outer=top
        fc_zbot = fc_entz
        fc_ztop = fc_entz + BLK_T
        ! Block 中心 = pole 軸在 outer face (top) 的 XY
        ! outer = entry + (BLK_T/sin(tilt))*axis
        fc_dt = BLK_T/sin(TILT_UP)
        fc_cx = fc_entx + fc_dt*fc_ax
        fc_cy = fc_enty + fc_dt*fc_ay
    *else
        ! Lower: pole 從側面穿過 block → block 垂直居中在 pole 軸上
        fc_zbot = fc_endz - BLK_T/2
        fc_ztop = fc_endz + BLK_T/2
        ! Block 中心 = pole 尾端 XY
        fc_cx = fc_endx
        fc_cy = fc_endy
    *endif

    ! XY 方向向量 (水平面, 朝 workspace = azim+180°)
    fc_twx = -cos(fc_azim)
    fc_twy = -sin(fc_azim)
    fc_sdx = sin(fc_azim)
    fc_sdy = -cos(fc_azim)

    ! --- Block Part 1: 8 角點 (global) ---
    *get, KP_NUM, KP, , NUM, MAX
    *if, KP_NUM, LT, 1, THEN
        KP_NUM = 0
    *endif
    fc_bkp = KP_NUM + 1
    NUMSTR, KP, fc_bkp

    ! Bottom face (z=fc_zbot), 4 corners: (s1a,u1a) (s1b,u1a) (s1b,u1b) (s1a,u1b)
    K,fc_bkp,  fc_cx+fc_s1a*fc_twx+fc_u1a*fc_sdx, fc_cy+fc_s1a*fc_twy+fc_u1a*fc_sdy, fc_zbot
    K,fc_bkp+1,fc_cx+fc_s1b*fc_twx+fc_u1a*fc_sdx, fc_cy+fc_s1b*fc_twy+fc_u1a*fc_sdy, fc_zbot
    K,fc_bkp+2,fc_cx+fc_s1b*fc_twx+fc_u1b*fc_sdx, fc_cy+fc_s1b*fc_twy+fc_u1b*fc_sdy, fc_zbot
    K,fc_bkp+3,fc_cx+fc_s1a*fc_twx+fc_u1b*fc_sdx, fc_cy+fc_s1a*fc_twy+fc_u1b*fc_sdy, fc_zbot
    ! Top face (z=fc_ztop)
    K,fc_bkp+4,fc_cx+fc_s1a*fc_twx+fc_u1a*fc_sdx, fc_cy+fc_s1a*fc_twy+fc_u1a*fc_sdy, fc_ztop
    K,fc_bkp+5,fc_cx+fc_s1b*fc_twx+fc_u1a*fc_sdx, fc_cy+fc_s1b*fc_twy+fc_u1a*fc_sdy, fc_ztop
    K,fc_bkp+6,fc_cx+fc_s1b*fc_twx+fc_u1b*fc_sdx, fc_cy+fc_s1b*fc_twy+fc_u1b*fc_sdy, fc_ztop
    K,fc_bkp+7,fc_cx+fc_s1a*fc_twx+fc_u1b*fc_sdx, fc_cy+fc_s1a*fc_twy+fc_u1b*fc_sdy, fc_ztop

    V,fc_bkp,fc_bkp+1,fc_bkp+2,fc_bkp+3,fc_bkp+4,fc_bkp+5,fc_bkp+6,fc_bkp+7

    ! --- Block Part 2: 8 角點 ---
    *get, KP_NUM, KP, , NUM, MAX
    fc_bkp2 = KP_NUM + 1
    NUMSTR, KP, fc_bkp2

    K,fc_bkp2,  fc_cx+fc_s2a*fc_twx+fc_u2a*fc_sdx, fc_cy+fc_s2a*fc_twy+fc_u2a*fc_sdy, fc_zbot
    K,fc_bkp2+1,fc_cx+fc_s2b*fc_twx+fc_u2a*fc_sdx, fc_cy+fc_s2b*fc_twy+fc_u2a*fc_sdy, fc_zbot
    K,fc_bkp2+2,fc_cx+fc_s2b*fc_twx+fc_u2b*fc_sdx, fc_cy+fc_s2b*fc_twy+fc_u2b*fc_sdy, fc_zbot
    K,fc_bkp2+3,fc_cx+fc_s2a*fc_twx+fc_u2b*fc_sdx, fc_cy+fc_s2a*fc_twy+fc_u2b*fc_sdy, fc_zbot
    K,fc_bkp2+4,fc_cx+fc_s2a*fc_twx+fc_u2a*fc_sdx, fc_cy+fc_s2a*fc_twy+fc_u2a*fc_sdy, fc_ztop
    K,fc_bkp2+5,fc_cx+fc_s2b*fc_twx+fc_u2a*fc_sdx, fc_cy+fc_s2b*fc_twy+fc_u2a*fc_sdy, fc_ztop
    K,fc_bkp2+6,fc_cx+fc_s2b*fc_twx+fc_u2b*fc_sdx, fc_cy+fc_s2b*fc_twy+fc_u2b*fc_sdy, fc_ztop
    K,fc_bkp2+7,fc_cx+fc_s2a*fc_twx+fc_u2b*fc_sdx, fc_cy+fc_s2a*fc_twy+fc_u2b*fc_sdy, fc_ztop

    V,fc_bkp2,fc_bkp2+1,fc_bkp2+2,fc_bkp2+3,fc_bkp2+4,fc_bkp2+5,fc_bkp2+6,fc_bkp2+7

    ! VADD
    *get, fc_bv2, VOLU, 0, NUM, MAX
    fc_bv1 = fc_bv2 - 1
    VADD, fc_bv1, fc_bv2

    ALLSEL, ALL

    *msg, INFO, fc_pi
     Block for P%I built

*enddo

! ===== [ADDED] 安裝 Yoke =====
! 環形板, 中心在原點, 軸 Z, 在上層 coil 頂面上方
! R_inner=40, R_outer=52, 厚度 2mm

YOKE_RI = 38.0*MM             ! [MODIFIED] 內徑縮小讓導柱不露出
YOKE_RO = 62.5*MM             ! [MODIFIED] 從導磁鐵環.STEP 提取
YOKE_T  = 2.0*MM

! 上層 coil top z: 需要從上層 block 計算
! 上層 entry_z 已在 coil 迴圈中算過, 重新算一次
! 所有上層 pole 的 fc_endz 相同, 取 P2 的
fc_up_endz = fc_tz(2) + POLE_TOTAL_LEN*sin(TILT_UP)
fc_up_entz = fc_up_endz - PENE_UP*sin(TILT_UP)
fc_up_blk_top = fc_up_entz + BLK_T
fc_yoke_zbot = fc_up_blk_top + BLK_YOKE_GAP   ! [MODIFIED] 用 gap 定位 yoke(不用 coil 高)
fc_yoke_ztop = fc_yoke_zbot + YOKE_T

! Yoke 截面 KP (矩形: R_inner~R_outer, zbot~ztop)
*get, KP_NUM, KP, , NUM, MAX
*if, KP_NUM, LT, 1, THEN
    KP_NUM = 0
*endif
fc_ykp = KP_NUM + 1
NUMSTR, KP, fc_ykp

K, fc_ykp,   YOKE_RI, 0, fc_yoke_zbot
K, fc_ykp+1, YOKE_RO, 0, fc_yoke_zbot
K, fc_ykp+2, YOKE_RO, 0, fc_yoke_ztop
K, fc_ykp+3, YOKE_RI, 0, fc_yoke_ztop
! 軸 KP (Z 軸, 通過原點)
K, fc_ykp+4, 0, 0, fc_yoke_zbot
K, fc_ykp+5, 0, 0, fc_yoke_ztop

*get, LI_NUM, LINE, , NUM, MAX
