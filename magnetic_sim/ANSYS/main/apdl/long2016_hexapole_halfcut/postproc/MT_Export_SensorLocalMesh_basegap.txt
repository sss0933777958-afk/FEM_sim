! ============================================================================
! MT_Export_SensorLocalMesh_basegap.txt
!   BASEGAP-mesh variant of MT_Export_SensorLocalMesh_graded.txt.
!   RESUME the REAL 200um air-slab gap mesh (db/geom/mesh_graded_basegap/mesh_graded.db,
!   mesh-only, NO solve) and dump the LOCAL air mesh connectivity around the 6
!   Hall-sensor centres so MATLAB can do true-FEM-tet barycentric interpolation of
!   the (already-exported) gap_200um nodal B (.dat, coilN/gap_200um) at sample
!   points inside each sensor cylinder.
!   Node IDs match the coilN/<gap variant> .dat because those solves RESUMEd THIS mesh
!   (db/geom/mesh_graded_basegap/mesh_graded.db, per MT_Sim_P*_basegap.txt).
!   [2026-07-12] db is now the 1.21M **gap_300um** conformal real-slab mesh; output
!     retargeted to csv/gap_300um/ (was csv/ for the old 706k gap_200um mesh, kept on disk).
!   ⚠ The graded CSV (data/mesh/graded/csv) is on a DIFFERENT mesh (636k) and is
!     mis-aligned (~20 mm) to gap_200um (706k) -> must use THIS basegap CSV instead.
!
!   Sensor centres = build_sensor_geometry (2026-07-10 upper-pole real-FEM-angle
!   inc_up=36.59deg; lower poles P1/P3/P6 unchanged, upper P2/P4/P5 shifted ~0.1 mm).
!
!   Output (CSV, absolute dir = ANSYS_data/long2016_hexapole_halfcut/csv/):
!     sensor_local_nodes.csv : nodeID, x, y, z         [m, ANSYS frame]
!     sensor_local_elems.csv : elemID, n1..n8          (SOLID96 8-node slots)
!   Consumed by extract_Vmat_interp.m (mesh_csv_dir = ANSYS_data/.../csv) for the
!   gap_200um (basegap) variant. No solve, no geometry change -- pure mesh export.
! ============================================================================

FINISH
/CLEAR,NOSTART
RESUME,'mesh_graded','db','G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\long2016_hexapole_halfcut\db\mesh\mesh_graded_basegap\'
/PREP7

! ---- 6 sensor centres (ANSYS frame [m]) = build_sensor_geometry (upper real FEM angle) ----
*DIM,SCX,ARRAY,6
*DIM,SCY,ARRAY,6
*DIM,SCZ,ARRAY,6
SCX(1)=  4.811056e-3, -3.169265e-3, -2.405528e-3,  1.584632e-3,  1.584632e-3, -2.405528e-3
SCY(1)=  0.000000e-3,  0.000000e-3,  4.166496e-3, -2.744664e-3,  2.744664e-3, -4.166496e-3
SCZ(1)= -14.298681e-3, -8.755489e-3,-14.298681e-3, -8.755489e-3, -8.755489e-3,-14.298681e-3
HB = 1.0e-3                       ! half-box [m] around each sensor

! ---- union of 6 node boxes ----
*do,i,1,6
    ALLSEL,ALL
    NSEL,S,LOC,X,SCX(i)-HB,SCX(i)+HB
    NSEL,R,LOC,Y,SCY(i)-HB,SCY(i)+HB
    NSEL,R,LOC,Z,SCZ(i)-HB,SCZ(i)+HB
    CM,SB%i%,NODE
*enddo
CMSEL,S,SB1
*do,i,2,6
    CMSEL,A,SB%i%
*enddo

! ---- elements touching those nodes, keep SOLID96, then full node set ----
ESLN,S,0                          ! any selected node -> element selected
ESEL,R,ENAME,,96                  ! SOLID96 only (drop any non-volume)
NSLE,S,ALL                        ! re-select ALL nodes of those elements (full tets)

! ---- dump nodes ----
*GET,NN,NODE,0,COUNT
*DIM,NID,ARRAY,NN
*DIM,NCX,ARRAY,NN
*DIM,NCY,ARRAY,NN
*DIM,NCZ,ARRAY,NN
ndd=0
*do,ii,1,NN
    ndd=NDNEXT(ndd)
    NID(ii)=ndd
    NCX(ii)=NX(ndd)
    NCY(ii)=NY(ndd)
    NCZ(ii)=NZ(ndd)
*enddo
/OUTPUT,'sensor_local_nodes','csv','G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\long2016_hexapole_halfcut\csv\gap_300um\'
*VWRITE,NID(1),NCX(1),NCY(1),NCZ(1)
(F12.0,',',ES23.14,',',ES23.14,',',ES23.14)
/OUTPUT

! ---- dump elements (8 node slots) ----
*GET,NE,ELEM,0,COUNT
*DIM,EID,ARRAY,NE
*DIM,EN,ARRAY,NE,8
edd=0
*do,ii,1,NE
    edd=ELNEXT(edd)
    EID(ii)=edd
    *do,k,1,8
        EN(ii,k)=NELEM(edd,k)
    *enddo
*enddo
/OUTPUT,'sensor_local_elems','csv','G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\long2016_hexapole_halfcut\csv\gap_300um\'
*VWRITE,EID(1),EN(1,1),EN(1,2),EN(1,3),EN(1,4),EN(1,5),EN(1,6),EN(1,7),EN(1,8)
(F12.0,8(',',F12.0))
/OUTPUT

/COM === SENSOR LOCAL MESH EXPORT (BASEGAP) DONE: NN=%NN% nodes, NE=%NE% SOLID96 elems ===
FINISH
