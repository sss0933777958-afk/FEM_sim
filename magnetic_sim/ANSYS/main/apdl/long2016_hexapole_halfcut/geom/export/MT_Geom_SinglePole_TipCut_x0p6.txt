! ====================================================================
! long2016 single-pole — FILLED POLE + yz-CUT at x=0.6mm  (IGES, inch-scaled for SolidWorks)

! magnetic pole (full cone, flat-tipped at x=0.6mm) + 4 support BLOCKs + protrusion (R5x7 mm), NO coil.
! Built in mm values, WP shifted to ORIGIN, then VLSCALE mm->inch (x1/25.4) so the
!   IGES (unit flag patched 6->1 = INCH) reads back as correct mm in SolidWorks
!   (ANSYS IGES has blank units-name -> SolidWorks defaults to INCH; same trick as hung).
! Output -> IGES/long_fei/singlepole_tipcut_x0p6.iges  (patched copy in model_check/long_fei/)
! ====================================================================
/CWD,'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\long2016_hexapole_halfcut\db\geom\geom_singlepole_variants'

Finish
/cle
/PREP7
*afun, DEG
BTOL, 1e-4
/UNITS, MKS

R_norm    = 0.5
R_norm_xy = R_norm*sqrt(2.0/3.0)
R_norm_z  = R_norm/sqrt(3.0)

YOKE_IN_R  = 84/2.0
YOKE_OUT_R = 106/2.0
YOKE_MID_R = (YOKE_IN_R + YOKE_OUT_R)/2.0
YOKE_H     = 2.0
PROT_R     = 10.0/2.0
PROT_H     = 7.0

POLE_R        = 6.0/2
POLE_TIP_R    = 40.0e-3
POLE_CONE_LEN = 15
POLE_CONE_HYP = sqrt(POLE_CONE_LEN**2 + POLE_R**2)
ANG1 = acos(POLE_R/POLE_CONE_HYP)
ANG2 = 90 - ANG1
L = POLE_CONE_LEN*POLE_TIP_R/POLE_R
POLE_TIP_CENTER = L/cos(ANG2)
POLE_CONE2_LEN  = L*cos(ANG2)
POLE_TIP_R2     = POLE_TIP_R*cos(ANG2)
POLE_TIP_FILLET_DROP = POLE_TIP_CENTER - POLE_TIP_R
POLE_TIP_FILLET_PROT = (POLE_CONE2_LEN - POLE_TIP_FILLET_DROP)

! ===== protrusion (coil core) =====
WPLANE, 1,0,0,0,,,,,,
WPcsys,-1,0
CYL4, YOKE_MID_R*cos(0), YOKE_MID_R*sin(0), PROT_R, ,,, -PROT_H

! ===== pole (azimuth 0): 4 support BLOCKs + cone =====
X0_LOW = YOKE_MID_R*cos(0)
Y0_LOW = YOKE_MID_R*sin(0)
Z0_LOW = -PROT_H
WPLANE, 1,0,0,0,,,,,,
WPcsys,-1,0
ROTA_ANG = 0
WPROTA, ROTA_ANG
*get, VL_NUM, VOLU, ,NUM, MAX
VL_START = VL_NUM + 1
numstr, VOLU, VL_START
BLOCK, X0_LOW - 5,  X0_LOW + 11, Y0_LOW - 5,  Y0_LOW + 5,  Z0_LOW - 10, Z0_LOW
BLOCK, X0_LOW - 5,  X0_LOW - 2,  Y0_LOW - 10, Y0_LOW + 10, Z0_LOW - 10, Z0_LOW - 5
BLOCK, X0_LOW - 10, X0_LOW - 2,  Y0_LOW - 5,  Y0_LOW + 5,  Z0_LOW - 10, Z0_LOW - 2
BLOCK, X0_LOW - 10, X0_LOW - 5,  Y0_LOW - 10, Y0_LOW + 10, Z0_LOW - 10, Z0_LOW - 5
*get, KP_NUM, KP  , ,NUM, MAX
*get, LI_NUM, LINE, ,NUM, MAX
KP_START = KP_NUM + 1
LI_START = LI_NUM + 1
numstr, KP,   KP_START
numstr, LINE, LI_START
l1 = R_norm_xy
l1_ang = atan2( 0 , R_norm_xy )
l2 = sqrt( (R_norm_xy + POLE_TIP_FILLET_PROT)**2 + POLE_TIP_R2**2 )
l2_ang = atan2( POLE_TIP_R2 , (R_norm_xy + POLE_TIP_FILLET_PROT) )
POLE_CONE_LEN_SHIFT = POLE_CONE_LEN + (R_norm_xy - POLE_TIP_FILLET_DROP)
l3 = sqrt( POLE_CONE_LEN_SHIFT**2 + POLE_R**2 )
l3_ang = atan2( POLE_R , POLE_CONE_LEN )
l4 = sqrt( (X0_LOW - 10)**2 + POLE_R**2 )
l4_ang = atan2( POLE_R , (X0_LOW - 10) )
l5 = abs( X0_LOW - 10 )
l5_ang = atan2( 0 , (X0_LOW - 10) )
l6 = abs( R_norm_xy + POLE_TIP_R )
l6_ang = atan2( 0, R_norm_xy + POLE_TIP_R )
K , , l1*cos(ROTA_ANG + l1_ang) , l1*sin(ROTA_ANG + l1_ang), Z0_LOW - 6
K , , l2*cos(ROTA_ANG + l2_ang) , l2*sin(ROTA_ANG + l2_ang), Z0_LOW - 6
K , , l3*cos(ROTA_ANG + l3_ang) , l3*sin(ROTA_ANG + l3_ang), Z0_LOW - 6
K , , l4*cos(ROTA_ANG + l4_ang) , l4*sin(ROTA_ANG + l4_ang), Z0_LOW - 6
K , , l5*cos(ROTA_ANG + l5_ang) , l5*sin(ROTA_ANG + l5_ang), Z0_LOW - 6
K , , l6*cos(ROTA_ANG + l6_ang) , l6*sin(ROTA_ANG + l6_ang), Z0_LOW - 6
LARC , KP_START , KP_START + 1 , KP_START + 5 , POLE_TIP_R
L , KP_START + 1 , KP_START + 2,,,,,,,,
L , KP_START + 2 , KP_START + 3,,,,,,,,
L , KP_START + 3 , KP_START + 4,,,,,,,,
L , KP_START + 4 , KP_START,,,,,,,,
*get, AR_NUM, AREA, ,NUM, MAX
AR_START = AR_NUM + 1
numstr, AREA, AR_START
AL, LI_START, LI_START + 1, LI_START + 2, LI_START + 3, LI_START + 4
VROTAT, AR_START, ,,,,,KP_START,KP_START + 4,,4

! ===== VADD all steel -> 1 vol =====
ALLSEL, ALL
VADD, ALL
NUMCMP, VOLU
*get, vol_steel, VOLU,,NUM,MAX

! ===== [MODIFIED] place pole on +x axis: tip apex -> (R_norm,0,0)=(500,0,0)um (pure translation, cone axis already +x) =====
X_SHIFT = R_norm - R_norm_xy
Z_SHIFT = -(Z0_LOW - 6)
ALLSEL, ALL
VGEN, 1, ALL, , , X_SHIFT, 0, Z_SHIFT, , , 1
ALLSEL, BELOW, VOLU
NUMCMP, VOLU
*get, vol_steel, VOLU,,NUM,MAX

! ===== [MODIFIED] TIP-CUT: yz-plane (perp x) at x=0.510mm. Tip apex now at x=0.5, body at x>0.5.
!        Cut, then delete the x<0.510 tip cap (10um flat), keep the x>0.510 body. =====
XCUT = 0.510
WPLANE, -1, XCUT,0,0, XCUT,1,0, XCUT,0,1
VSBW, vol_steel
WPCSYS, -1, 0
CSYS, 0
ALLSEL, ALL
VSEL, S, LOC, X, -100, XCUT
VDELE, ALL, , , 1
ALLSEL, ALL
NUMCMP, VOLU

! ===== diagnostics (flat tip now at x=0.510) =====
ALLSEL, BELOW, VOLU
*GET, XKMIN, KP, 0, MNLOC, X
*GET, XKMAX, KP, 0, MXLOC, X
/GO
*MSG, INFO, XKMIN, XKMAX
 singlepole_tipcut: KP x range [%G , %G] mm (flat tip expect x-min=0.510)
*get, fc_nvf, VOLU, 0, NUM, MAX
*msg, INFO, fc_nvf
 singlepole_tipcut: %I volumes

! ===== [MODIFIED] render PNG (visual review: full iso, y=0 side, tip zoom) =====
/SHOW, PNG
/PNUM, VOLU, 0

/ANG, 1

/ANG, 1
/AUTO, 1

! ===== [MODIFIED] scale mm -> inch (x1/25.4) for SolidWorks (SW ignores IGES unit flag, defaults to
!        INCH); flag patched 6->1 AFTER run so SW measures correct mm. Same trick as hung. =====
INCH = 1.0/25.4
ALLSEL, BELOW, VOLU
VLSCALE, ALL, , , INCH, INCH, INCH, , , 1
ALLSEL, BELOW, VOLU
IGESOUT, 'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\model_check\long_fei\singlepole_tipcut', 'iges'
FINISH
/EXIT, NOSAV
