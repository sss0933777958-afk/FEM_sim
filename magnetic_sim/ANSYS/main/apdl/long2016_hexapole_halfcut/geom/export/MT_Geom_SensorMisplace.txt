PLO(1) = 1,   0,   1,   0,  0,   1        ! 1=下極(P1/P3/P6)、0=上極
*get, VG0, VOLU,, NUM, MAX
*do, ip, 1, 6
	th = PTH(ip)
	*if, PLO(ip), EQ, 1, then              ! 下極：錐軸水平、環 φ -90..+90
		elev = -PSI_S
		axx = cos(th)            $ axy = sin(th)            $ axz = 0
		e2a = -BETA_S
		MLO = -3 $ MHI = 3
		ABASEP = ABASE_LO
	*else                                  ! 上極：錐軸傾 INC、環 φ 0..330
		elev = +PSI_S
		axx = cos(INC_S)*cos(th) $ axy = cos(INC_S)*sin(th) $ axz = sin(INC_S)
		e2a = INC_S + BETA_S
		MLO = 0 $ MHI = 11
		ABASEP = ABASE_HI
	*endif
	! 尖端(全域) = R_norm·dir(elev,θ) + [0,0,SPH_OFST]
	tpx = RN_S*cos(elev)*cos(th)
	tpy = RN_S*cos(elev)*sin(th)
	tpz = RN_S*sin(elev) - 13e-3 + RN_S/sqrt(3.0)
	! φ=0 slant e2、û=(e2−cosβ·â)/sinβ、v̂=â×û
	e2x = cos(e2a)*cos(th) $ e2y = cos(e2a)*sin(th) $ e2z = sin(e2a)
	uux = (e2x - CB*axx)/SB $ uuy = (e2y - CB*axy)/SB $ uuz = (e2z - CB*axz)/SB
	vvx = axy*uuz - axz*uuy
	vvy = axz*uux - axx*uuz
	vvz = axx*uuy - axy*uux
	! --- 前後線 φ=0：soff=SOFF0+k·STEP_FA, k=-2..10（ê2(0)/n̂(0) 固定）---
	ex = CB*axx + SB*uux  $ ey = CB*axy + SB*uuy  $ ez = CB*axz + SB*uuz
	nx = -SB*axx + CB*uux $ ny = -SB*axy + CB*uuy $ nz = -SB*axz + CB*uuz
	*do, kk, -2, 10
		soff = SOFF0 + kk*STEP_FA
		scx = tpx + soff*ex + OFFN*nx
		scy = tpy + soff*ey + OFFN*ny
		scz = tpz + soff*ez + OFFN*nz
		WPLANE,1,0,0,0 $ WPCSYS,-1,0 $ WPOFFS, scx, scy, scz
		SPHERE, 0, SPH_R, 0, 360
	*enddo
	! --- shank 段前後（a=15..ABASEP mm, 步1mm；shank 面軸平行 → n̂=徑向 û；球心離軸 RSHANK+OFFN）---
	*do, aa, 15, ABASEP
		am = aa*1e-3
		scx = tpx + am*axx + (RSHANK+OFFN)*uux
		scy = tpy + am*axy + (RSHANK+OFFN)*uuy
		scz = tpz + am*axz + (RSHANK+OFFN)*uuz
		WPLANE,1,0,0,0 $ WPCSYS,-1,0 $ WPOFFS, scx, scy, scz
		SPHERE, 0, SPH_R, 0, 360
	*enddo
	! --- 方位環 soff=SOFF0：φ=m·STEP_AZ, m=MLO..MHI 略 m=0（w=cosφ û+sinφ v̂）---
	*do, mm, MLO, MHI
		*if, mm, NE, 0, then
			phi = mm*STEP_AZ
			wx = cos(phi)*uux + sin(phi)*vvx
			wy = cos(phi)*uuy + sin(phi)*vvy
			wz = cos(phi)*uuz + sin(phi)*vvz
			ex2 = CB*axx + SB*wx  $ ey2 = CB*axy + SB*wy  $ ez2 = CB*axz + SB*wz
			nx2 = -SB*axx + CB*wx $ ny2 = -SB*axy + CB*wy $ nz2 = -SB*axz + CB*wz
			scx = tpx + SOFF0*ex2 + OFFN*nx2
			scy = tpy + SOFF0*ey2 + OFFN*ny2
			scz = tpz + SOFF0*ez2 + OFFN*nz2
			WPLANE,1,0,0,0 $ WPCSYS,-1,0 $ WPOFFS, scx, scy, scz
			SPHERE, 0, SPH_R, 0, 360
		*endif
	*enddo
*enddo
CSYS,0 $ WPCSYS,-1,0
*get, VG1, VOLU,, NUM, MAX
VSEL,S,VOLU,, VG0+1, VG1
CM, SPH_GRID, VOLU
ALLSEL,ALL
