! ============================================================================
! NTU_hexapole — UPPER-layer FULL ASSEMBLY (upper_assembly): 3 磁極 + 6 導柱 + yoke
! ----------------------------------------------------------------------------
! Source CAD = CAD_model/NTU_hexapole/STEP/建棋模擬_上層/{上層組合.STEP, YOKE.step}.
!   Canonical frame via T = Rz(180 about world z)+z-9.55 (pole0 canonical, == pole_assembly).
!   Dims OCC-extracted from the CAD (under T):
!     3 磁極 : flat 0.25mm, 9-seg profile, at 0/120/240 deg.
!     6 導柱 : SOLID stepped cyl, radius 20.5 at 0/60/120/180/240/300 deg; step at z=11.25.
!              base z: 0/120/240deg -> 0.25 ; 60/180/300deg -> -0.5 ; top z 17.25.
!              r2.5 (base->11.25) + r2.49 (11.25->17.25).
!     yoke   : rounded-rect plate 59.8x53.8 (r5 corners) + central rounded-rect cutout
!              31x20.5 (r5) + 6 post-holes r2.5 @radius20.5 ; thickness 5.5 @z[11.25,16.75].
!              deck uses SHARP rectangles (r5 corner fillets omitted, minor) per STEP dims.
!   STEP delivery (exact CAD) = make_upper_assembly_step.py -> upper_assembly.step.
!   model_check gets STEP only (no IGES) per user; this deck's IGESOUT goes to IGES/.
! [MODIFIED 2026-07-11] 3-pole -> 3-pole + 6-post + yoke.
! ============================================================================

/CWD,'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\ANSYS_data\NTU_hexapole\db\geom\geom_export'

Finish
/cle
/PREP7
*afun, DEG

UNIT_MM = 1
*if, UNIT_MM, eq, 1, then
    SC = 1.0
    BTOL, 1e-4
*else
    SC = 1.0e-3
    BTOL, 1e-7
*endif
/UNITS, MKS

TH    = 0.25*SC
POSTX = 20.5*SC
PR    = 20.5*SC          ! post/hole axis radius

! ============================================================================
! PART A — 3 磁極（WPROTA 0/120/240 deg）
! ============================================================================
*do, pole, 1, 3
	ANG = (pole-1)*120
	WPLANE, 1,0,0,0
	WPcsys,-1,0
	WPROTA, ANG
	CSYS, 4
	*get, KB, KP, , NUM, MAX
	K, KB+1,  0.357*SC,  0.005*SC, 0
	K, KB+2,  0.353*SC,  0.000*SC, 0
	K, KB+3,  0.357*SC, -0.005*SC, 0
	K, KB+4,  4.509*SC, -0.523*SC, 0
	K, KB+5,  6.175*SC, -0.877*SC, 0
	K, KB+6, 19.048*SC, -4.784*SC, 0
	K, KB+7, 19.048*SC,  4.784*SC, 0
	K, KB+8,  6.175*SC,  0.877*SC, 0
	K, KB+9,  4.509*SC,  0.523*SC, 0
	K, KB+10,25.500*SC,  0.000*SC, 0
	K, KB+11, 0.358*SC,  0.000*SC, 0
	K, KB+12, 3.271*SC,-10.446*SC, 0
	K, KB+13, 3.271*SC, 10.446*SC, 0
	K, KB+14,20.500*SC,  0.000*SC, 0
	*get, LB, LINE, , NUM, MAX
	LARC, KB+1, KB+2, KB+11, 0.005*SC
	LARC, KB+2, KB+3, KB+11, 0.005*SC
	L,    KB+3, KB+4
	LARC, KB+4, KB+5, KB+12, 10.0*SC
	L,    KB+5, KB+6
	LARC, KB+6, KB+10, KB+14, 5.0*SC
	LARC, KB+10, KB+7, KB+14, 5.0*SC
	L,    KB+7, KB+8
	LARC, KB+8, KB+9, KB+13, 10.0*SC
	L,    KB+9, KB+1
	AL, LB+1, LB+2, LB+3, LB+4, LB+5, LB+6, LB+7, LB+8, LB+9, LB+10
	*get, AB, AREA, , NUM, MAX
	VEXT, AB, , , 0, 0, TH
	KDELE, KB+11, KB+14, 1
	CSYS, 0
*enddo

! ============================================================================
! PART B — 6 導柱（radius 20.5，角 0/60/120/180/240/300；base z 交替 0.25 / -0.5）
! ============================================================================
*dim, PANG, , 6
PANG(1)=0 $ PANG(2)=60 $ PANG(3)=120 $ PANG(4)=180 $ PANG(5)=240 $ PANG(6)=300
*dim, PBZ, , 6
PBZ(1)=0.25 $ PBZ(2)=-0.5 $ PBZ(3)=0.25 $ PBZ(4)=-0.5 $ PBZ(5)=0.25 $ PBZ(6)=-0.5
*do, ip, 1, 6
	PX = PR*cos(PANG(ip))
	PY = PR*sin(PANG(ip))
	BZ = PBZ(ip)*SC
	WPLANE, 1,0,0,0
	WPcsys,-1,0
	WPOFFS, PX, PY, 0
	CYLIND, 2.50*SC, , BZ,        11.25*SC     ! r2.5  base -> 11.25
	CYLIND, 2.49*SC, , 11.25*SC,  17.25*SC     ! r2.49 11.25 -> 17.25
	WPOFFS, -PX, -PY, 0
*enddo

! ============================================================================
! PART C — yoke（外盤 − 中心 cutout − 6 孔；厚 5.5 @z[11.25,16.75]）
! ============================================================================
CSYS, 0
WPLANE, 1,0,0,0
WPcsys,-1,0
BLOCK, -29.900*SC, 29.900*SC, -26.900*SC, 26.900*SC, 11.25*SC, 16.75*SC   ! 外盤(sharp) = V_OUT
*get, V_OUT, VOLU, , NUM, MAX
BLOCK, -15.500*SC, 15.500*SC, -10.254*SC, 10.254*SC, 11.25*SC, 16.75*SC   ! 中心 cutout = V_OUT+1
*do, ih, 1, 6                                  ! 6 導柱孔 = V_OUT+2 .. V_OUT+7（先全建、序號連續）
	HX = PR*cos(PANG(ih))
	HY = PR*sin(PANG(ih))
	WPLANE, 1,0,0,0
	WPcsys,-1,0
	WPOFFS, HX, HY, 0
	CYLIND, 2.50*SC, , 11.25*SC, 16.75*SC      ! 導柱孔 r2.5
	WPOFFS, -HX, -HY, 0
*enddo
! 一次 VSBV：外盤 − (cutout + 6 孔)；選 V_OUT..V_OUT+7 後 VSBV ALL（避免序號重用踩坑）
VSEL, S, VOLU, , V_OUT, V_OUT+7
VSBV, V_OUT, ALL
ALLSEL, ALL

! ---- finalise + IGES（→ IGES/，不進 model_check）---------------------------
ALLSEL, ALL
NUMCMP, ALL

*if, UNIT_MM, eq, 1, then
    IGESOUT, 'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\IGES\NTU_hexapole\upper_assembly_mm', 'iges'
*else
    IGESOUT, 'G:\my_workspace\code\FEM_sim\magnetic_sim\ANSYS\main\IGES\NTU_hexapole\upper_assembly', 'iges'
*endif

SAVE
